\subsection{Skew Matrix}
Vector space isomorphism where $r \in \mathbb{R}^3$
\begin{equation}
    S(r)
=
\begin{bmatrix}
0 & -r_3 & r_2 \\
r_3 & 0 & -r_1 \\
-r_2 & r_1 & 0
\end{bmatrix},
\qquad
S^{-1}\left(S(r)\right) = r
\tag{A1}
\label{annex:skew_matrix}
\end{equation}
